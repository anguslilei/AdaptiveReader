import Foundation
import CryptoKit

@main struct Feature182Probe {
    static func main() async {
        let url = URL(fileURLWithPath: CommandLine.arguments[1])
        do {
            let package = try await EPUBSemanticPackageLoader.load(fileURL: url)
            print("PACKAGE archive=\(package.archiveSHA256) spine=\(package.spine.count)")
            let reader = try await EPUBSemanticResourceReader.open(fileURL: url, expectedArchiveSHA256: package.archiveSHA256)
            do {
                for occurrence in [0, 1, 3, 8, 20] where occurrence < package.spine.count {
                    let entry = package.spine[occurrence]
                    let path = package.manifestItems[entry.manifestIndex].path
                    let resource = try await reader.read(path: path)
                    let start = ContinuousClock.now
                    do {
                        let s = try await SemanticXHTMLExtractor.extract(resource: resource, spineOccurrence: occurrence)
                        let tree = try await SemanticXMLParser.parse(resource.bytes)
                        var count = 0
                        for block in s.blocks {
                            for run in block.runs {
                                var index = tree.rootIndex
                                for slot in run.anchor.nodePath.components { index = tree.nodes[index].children[slot] }
                                guard case .text(let text) = tree.nodes[index].kind,
                                      text.utf8.elementsEqual(run.text.utf8),
                                      run.anchor.textRange?.upperBound == text.utf16.count else { fatalError("source mismatch") }
                                count += 1
                            }
                        }
                        print("CHAPTER occurrence=\(occurrence) path=\(path) bytes=\(resource.bytes.count) sha256=\(resource.sha256) blocks=\(s.blocks.count) runs=\(count) opaque=\(s.blocks.filter{$0.role == .opaque}.count) duration=\(start.duration(to:.now)) verified=true")
                    } catch { print("CHAPTER occurrence=\(occurrence) path=\(path) rejected=\(error)") }
                }
                await reader.close()
            } catch { await reader.close(); throw error }
        } catch { print("PACKAGE rejected=\(error)") }
    }
}
